função plusOne(digits):
    n ← tamanho(digits)
    
    // Percorre do último dígito até o primeiro
    para i de n - 1 até 0 passo -1 faça:
        se digits[i] < 9 então:
            digits[i] ← digits[i] + 1
            retornar digits  // Soma concluída sem propagação extra de "vai um"
        fim se
        
        digits[i] ← 0  // Se o dígito era 9, vira 0 e propagamos o "vai um"
    fim para
    
    // Se o loop terminou, todos os dígitos eram 9 (ex: [9, 9] -> [0, 0])
    // Precisamos criar um novo vetor com tamanho n + 1 e colocar 1 na frente
    novoArray ← array de tamanho (n + 1) com zeros
    novoArray[0] ← 1
    
    retornar novoArray
